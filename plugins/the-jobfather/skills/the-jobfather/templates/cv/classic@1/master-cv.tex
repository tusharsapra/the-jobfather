% Organisation/date and title/location use two compact, baseline-aligned rows.
\newcommand{\resumeEntry}[4]{%
  \begin{tabular*}{\textwidth}{@{\extracolsep{\fill}}lr@{}}
    \textbf{\fontsize{11}{12.5}\selectfont #1} &
    {\color{secondary}\fontsize{10.2}{11.5}\selectfont #2} \\[1.2pt]
    {\color{secondary}\textit{\fontsize{10.4}{11.8}\selectfont #3}} &
    {\color{secondary}\textit{\fontsize{10.2}{11.5}\selectfont #4}}
  \end{tabular*}\par
}

% A heading-only entry (for example internships or freelance work) with dates.
\newcommand{\resumeGroup}[2]{%
  \begin{tabular*}{\textwidth}{@{\extracolsep{\fill}}lr@{}}
    \textbf{\fontsize{11}{12.5}\selectfont #1} &
    {\color{secondary}\fontsize{10.2}{11.5}\selectfont #2}
  \end{tabular*}\par
}
